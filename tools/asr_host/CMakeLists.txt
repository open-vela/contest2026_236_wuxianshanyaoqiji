cmake_minimum_required(VERSION 3.16)
project(qiji_asr_host C)
if(NOT DEFINED OFFICIAL_ROOT OR NOT DEFINED MBEDTLS_BUILD)
  message(FATAL_ERROR "Set OFFICIAL_ROOT and MBEDTLS_BUILD")
endif()
set(APP ${CMAKE_CURRENT_LIST_DIR}/../../app/gemini_chat_minimal)
set(MBED ${OFFICIAL_ROOT}/apps/crypto/mbedtls/mbedtls)
set(CJSON ${OFFICIAL_ROOT}/apps/netutils/cjson/cJSON)
add_executable(qiji_asr_host main.c ${APP}/qiji_aliyun_asr.c ${CJSON}/cJSON.c)
target_include_directories(qiji_asr_host PRIVATE stubs ${APP} ${CJSON} ${MBED}/include ${OFFICIAL_ROOT}/packages/ai_agent/include)
target_compile_options(qiji_asr_host PRIVATE -Wall -Wextra -Werror)
target_link_libraries(qiji_asr_host PRIVATE ${MBEDTLS_BUILD}/library/libmbedtls.a ${MBEDTLS_BUILD}/library/libmbedx509.a ${MBEDTLS_BUILD}/library/libmbedcrypto.a m)
add_executable(qiji_playback_test test_playback.c ${OFFICIAL_ROOT}/packages/ai_agent/src/voice/audio_playback.c)
target_include_directories(qiji_playback_test PRIVATE stubs ${OFFICIAL_ROOT}/packages/ai_agent/include)
target_compile_options(qiji_playback_test PRIVATE -Wall -Wextra -Werror)
file(GLOB PFW_SOURCES ${OFFICIAL_ROOT}/frameworks/multimedia/media/pfw/*.c)
add_executable(qiji_policy_test test_policy.c ${PFW_SOURCES})
target_include_directories(qiji_policy_test PRIVATE ${OFFICIAL_ROOT}/frameworks/multimedia/media/pfw/include)
target_compile_definitions(qiji_policy_test PRIVATE CONFIG_LIB_PFW_DEBUG)

target_link_libraries(qiji_playback_test PRIVATE pthread)
