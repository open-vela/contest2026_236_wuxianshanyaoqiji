cmake_minimum_required(VERSION 3.16)
project(qiji_portrait_preview C CXX ASM)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
add_subdirectory(${OFFICIAL_ROOT}/external/freetype/freetype freetype)
set(LVGL_SOURCE ${OFFICIAL_ROOT}/apps/graphics/lvgl/lvgl)
add_subdirectory(${LVGL_SOURCE} lvgl)
target_link_libraries(lvgl PUBLIC freetype)
add_executable(portrait_preview main.c
    ../../app/gemini_chat_minimal/qiji_avatar.c
    ../../app/gemini_chat_minimal/qiji_sprite.c
    ../../app/gemini_chat_minimal/qiji_layers.c
    ../../app/gemini_chat_minimal/qiji_view.c)
target_include_directories(portrait_preview PRIVATE
    ${OFFICIAL_ROOT}/apps/graphics/lvgl ../../app/gemini_chat_minimal)
target_link_libraries(portrait_preview PRIVATE lvgl m)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
target_compile_definitions(portrait_preview PRIVATE QIJI_SPRITE_DIR="${PROJECT_ROOT}/assets/kotone-v1/bin")
target_compile_definitions(portrait_preview PRIVATE QIJI_LAYER_DIR="${PROJECT_ROOT}/assets/kotone-layered-v1/bin")
add_executable(touch_test touch_test.c)
enable_testing()
add_test(NAME portrait_touch COMMAND touch_test)
add_executable(key_test key_test.c)
add_test(NAME onboard_enter COMMAND key_test)
add_executable(scroll_test scroll_test.c)
add_test(NAME speech_scroll COMMAND scroll_test)
add_executable(dialogue_test dialogue_test.c)
add_test(NAME spoken_dialogue COMMAND dialogue_test)
add_executable(envelope_test envelope_test.c)
add_test(NAME pcm_envelope COMMAND envelope_test)
foreach(mode rig fallback)
    add_executable(layers_${mode}_test layers_test.c
        ../../app/gemini_chat_minimal/qiji_sprite.c
        ../../app/gemini_chat_minimal/qiji_layers.c)
    target_include_directories(layers_${mode}_test PRIVATE
        ${OFFICIAL_ROOT}/apps/graphics/lvgl ../../app/gemini_chat_minimal)
    target_link_libraries(layers_${mode}_test PRIVATE lvgl m)
    target_compile_definitions(layers_${mode}_test PRIVATE QIJI_SPRITE_DIR="${PROJECT_ROOT}/assets/kotone-v1/bin")
    if(mode STREQUAL "rig")
        target_compile_definitions(layers_${mode}_test PRIVATE QIJI_LAYER_DIR="${PROJECT_ROOT}/assets/kotone-layered-v1/bin")
    else()
        target_compile_definitions(layers_${mode}_test PRIVATE QIJI_LAYER_DIR="/nonexistent-qiji-test" TEST_FALLBACK=1)
    endif()
    add_test(NAME layers_${mode} COMMAND layers_${mode}_test)
endforeach()
